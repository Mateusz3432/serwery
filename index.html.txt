    <script>
        // Tutaj wpisz adresy IP serwerów, których status chcesz pokazywać na żywo
        const servers = [
            "play.hypixel.net",
            "mc.survival.pl"
        ];

        async function loadServers() {
            const container = document.getElementById("servers-list");
            container.innerHTML = "";

            for (let ip of servers) {
                try {
                    let response = await fetch(`https://api.mcsrvstat.us/2/${ip}`);
                    let data = await response.json();

                    let isOnline = data.online;
                    let playersOnline = isOnline ? data.players.online : 0;
                    let playersMax = isOnline ? data.players.max : 0;

                    let card = document.createElement("div");
                    card.className = `server-card-status ${isOnline ? '' : 'offline'}`;
                    
                    card.innerHTML = `
                        <div class="server-info">
                            <h4>${ip}</h4>
                            <p>Gracze online: <b>${playersOnline} / ${playersMax}</b></p>
                        </div>
                        <div class="status-indicator">
                            ${isOnline ? '<span class="online-text">● ONLINE</span>' : '<span class="offline-text">● OFFLINE</span>'}
                        </div>
                    `;
                    container.appendChild(card);
                } catch (e) {
                    console.error("Błąd ładowania serwera", ip);
                }
            }
        }

        loadServers();
        setInterval(loadServers, 60000);
    </script>
